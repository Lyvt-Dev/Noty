cmake_minimum_required(VERSION 3.24)
project(Noty VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
set(wxBUILD_TESTS OFF CACHE BOOL "" FORCE)
set(wxBUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(wxUSE_WEBREQUEST ON CACHE BOOL "" FORCE)
FetchContent_Declare(wxWidgets GIT_REPOSITORY https://github.com/wxWidgets/wxWidgets.git GIT_TAG v3.2.8.1 GIT_SHALLOW TRUE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_Declare(picosha2 GIT_REPOSITORY https://github.com/okdshin/PicoSHA2.git GIT_TAG v1.0.1 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(wxWidgets json picosha2)
add_executable(Noty WIN32 MACOSX_BUNDLE src/main.cpp src/updater.cpp)
target_include_directories(Noty PRIVATE ${picosha2_SOURCE_DIR})
target_link_libraries(Noty PRIVATE wx::core wx::base wx::stc wx::aui wx::net nlohmann_json::nlohmann_json)
target_compile_definitions(Noty PRIVATE NOTY_VERSION="${PROJECT_VERSION}")
if(WIN32)
  target_link_libraries(Noty PRIVATE dwmapi)
  if(MSVC)
    target_compile_options(Noty PRIVATE /utf-8 /W4)
  endif()
endif()
set_target_properties(Noty PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME Noty MACOSX_BUNDLE_GUI_IDENTIFIER dev.lyvt.noty MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
install(TARGETS Noty BUNDLE DESTINATION . RUNTIME DESTINATION bin)
set(CPACK_PACKAGE_NAME Noty)
set(CPACK_PACKAGE_VENDOR "Lyvt-Dev")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
if(WIN32)
  set(CPACK_GENERATOR NSIS ZIP)
  set(CPACK_NSIS_DISPLAY_NAME "Noty")
  set(CPACK_NSIS_PACKAGE_NAME "Noty")
  set(CPACK_NSIS_EXECUTABLES_DIRECTORY bin)
  set(CPACK_PACKAGE_EXECUTABLES "Noty;Noty")
  set(CPACK_NSIS_INSTALLED_ICON_NAME "bin\\\\Noty.exe")
  set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
elseif(APPLE)
  set(CPACK_GENERATOR DragNDrop)
else()
  set(CPACK_GENERATOR DEB TGZ)
  set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Lyvt-Dev")
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  install(FILES packaging/noty.desktop DESTINATION share/applications)
endif()
include(CPack)

include(CTest)
if(BUILD_TESTING)
  add_executable(update_test tests/update_test.cpp)
  target_include_directories(update_test PRIVATE src ${picosha2_SOURCE_DIR})
  add_test(NAME update_validation COMMAND update_test)
endif()
