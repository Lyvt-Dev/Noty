cmake_minimum_required(VERSION 3.24)
project(Noty VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/MP2)
endif()
include(FetchContent)
set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
set(wxBUILD_INSTALL OFF CACHE BOOL "" FORCE)
set(wxBUILD_TESTS OFF CACHE BOOL "" FORCE)
set(wxBUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(wxUSE_WEBREQUEST ON CACHE BOOL "" FORCE)
FetchContent_Declare(wxWidgets GIT_REPOSITORY https://github.com/wxWidgets/wxWidgets.git GIT_TAG v3.2.8.1 GIT_SHALLOW TRUE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_Declare(picosha2 GIT_REPOSITORY https://github.com/okdshin/PicoSHA2.git GIT_TAG v1.0.1 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(wxWidgets json picosha2)
add_executable(Noty WIN32 MACOSX_BUNDLE src/main.cpp src/frame.cpp src/chrome.cpp src/document.cpp src/search.cpp src/updater.cpp)
target_include_directories(Noty PRIVATE ${picosha2_SOURCE_DIR})
target_link_libraries(Noty PRIVATE wx::core wx::base wx::stc wx::aui wx::net nlohmann_json::nlohmann_json)
target_compile_definitions(Noty PRIVATE NOTY_VERSION="${PROJECT_VERSION}")
if(WIN32)
  enable_language(RC)
  target_sources(Noty PRIVATE assets/noty.rc)
  target_include_directories(Noty PRIVATE assets)
  target_link_libraries(Noty PRIVATE dwmapi)
  if(MSVC)
    target_compile_options(Noty PRIVATE /utf-8 /W4)
    target_link_options(Noty PRIVATE /MANIFEST:NO)
  endif()
endif()
if(APPLE)
  target_sources(Noty PRIVATE assets/noty.icns)
  set_source_files_properties(assets/noty.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
endif()
set_target_properties(Noty PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME Noty MACOSX_BUNDLE_GUI_IDENTIFIER dev.lyvt.noty MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION} MACOSX_BUNDLE_ICON_FILE noty.icns)
add_executable(NotyUpdate src/update_helper.cpp)
target_include_directories(NotyUpdate PRIVATE ${picosha2_SOURCE_DIR})
if(WIN32)
  target_link_libraries(NotyUpdate PRIVATE shell32)
  if(MSVC)
    target_compile_options(NotyUpdate PRIVATE /utf-8 /W4)
  endif()
endif()
include(InstallRequiredSystemLibraries)
install(TARGETS Noty BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(APPLE)
  install(TARGETS NotyUpdate RUNTIME DESTINATION Noty.app/Contents/MacOS)
else()
  install(TARGETS NotyUpdate RUNTIME DESTINATION bin)
endif()
set(CPACK_PACKAGE_NAME Noty)
set(CPACK_PACKAGE_VENDOR "Lyvt-Dev")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_INSTALL_DIRECTORY Noty)
if(WIN32)
  set(CPACK_GENERATOR NSIS ZIP)
  set(CPACK_NSIS_DISPLAY_NAME "Noty")
  set(CPACK_NSIS_PACKAGE_NAME "Noty")
  set(CPACK_NSIS_EXECUTABLES_DIRECTORY bin)
  set(CPACK_PACKAGE_EXECUTABLES "Noty;Noty")
  set(CPACK_NSIS_INSTALLED_ICON_NAME "bin\\Noty.exe")
  set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL OFF)
  set(CPACK_NSIS_MUI_ICON "${CMAKE_CURRENT_SOURCE_DIR}/assets/noty.ico")
  set(CPACK_NSIS_MUI_UNIICON "${CMAKE_CURRENT_SOURCE_DIR}/assets/noty.ico")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/packaging/upgrade.nsh" CPACK_NSIS_DEFINES)
  set(CPACK_NSIS_EXTRA_PREINSTALL_COMMANDS "Call NotyExistingInstall\nSetOutPath \"$INSTDIR\"")
  set(CPACK_NSIS_EXTRA_INSTALL_COMMANDS "Call NotyRemoveLegacyEntry")
elseif(APPLE)
  set(CPACK_GENERATOR DragNDrop)
else()
  set(CPACK_GENERATOR DEB TGZ)
  set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Lyvt-Dev")
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  install(FILES packaging/noty.desktop DESTINATION share/applications)
  install(FILES assets/noty-logo.png DESTINATION share/icons/hicolor/256x256/apps RENAME noty.png)
endif()
set(NOTY_LICENSE_DIR share/Noty/licenses)
set(NOTY_EXAMPLE_DIR share/Noty/examples)
if(APPLE)
  set(NOTY_LICENSE_DIR Noty.app/Contents/Resources/licenses)
  set(NOTY_EXAMPLE_DIR Noty.app/Contents/Resources/examples)
endif()
install(FILES ${wxwidgets_SOURCE_DIR}/docs/licence.txt DESTINATION ${NOTY_LICENSE_DIR} RENAME wxWidgets.txt)
install(FILES ${json_SOURCE_DIR}/LICENSE.MIT DESTINATION ${NOTY_LICENSE_DIR} RENAME nlohmann-json.txt)
install(FILES ${picosha2_SOURCE_DIR}/LICENSE DESTINATION ${NOTY_LICENSE_DIR} RENAME PicoSHA2.txt)
set(CPACK_VERBATIM_VARIABLES ON)
include(CPack)

include(CTest)
if(BUILD_TESTING)
  add_test(NAME update_helper_arguments COMMAND NotyUpdate --self-test)
  add_executable(update_test tests/update_test.cpp)
  target_include_directories(update_test PRIVATE src ${picosha2_SOURCE_DIR})
  add_test(NAME update_validation COMMAND update_test)
  add_executable(codec_test tests/codec_test.cpp)
  target_include_directories(codec_test PRIVATE src)
  if(MSVC)
    target_compile_options(codec_test PRIVATE /utf-8)
  endif()
  add_test(NAME encoding_roundtrip COMMAND codec_test)
  add_executable(ui_text_test tests/ui_text_test.cpp)
  target_include_directories(ui_text_test PRIVATE src)
  target_link_libraries(ui_text_test PRIVATE wx::base)
  if(MSVC)
    target_compile_options(ui_text_test PRIVATE /utf-8)
  endif()
  add_test(NAME german_ui_labels COMMAND ui_text_test)
  add_executable(search_test tests/search_test.cpp src/search.cpp)
  target_include_directories(search_test PRIVATE src)
  target_link_libraries(search_test PRIVATE wx::base)
  if(MSVC)
    target_compile_options(search_test PRIVATE /utf-8)
  endif()
  add_test(NAME unicode_regex COMMAND search_test)
endif()
add_library(noty_example_plugin MODULE plugins/example.cpp)
target_include_directories(noty_example_plugin PRIVATE src)
install(TARGETS noty_example_plugin LIBRARY DESTINATION ${NOTY_EXAMPLE_DIR} RUNTIME DESTINATION ${NOTY_EXAMPLE_DIR})
install(FILES src/plugin_api.h DESTINATION ${NOTY_EXAMPLE_DIR})
